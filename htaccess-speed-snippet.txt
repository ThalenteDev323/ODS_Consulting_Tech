# ODS Consulting - speed settings (OPTIONAL, for domains.co.za)
# Add these lines to the END of the existing .htaccess file in public_html.
# Do not replace the file: it may already contain other rules. Take a copy first.

# Remember images, CSS and JavaScript in the visitor's browser for longer
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType image/jpeg "access plus 1 year"
  ExpiresByType image/png "access plus 1 year"
  ExpiresByType image/webp "access plus 1 year"
  ExpiresByType video/mp4 "access plus 1 year"
  ExpiresByType text/css "access plus 1 month"
  ExpiresByType application/javascript "access plus 1 month"
  ExpiresByType text/javascript "access plus 1 month"
</IfModule>

# Compress text files before sending them
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript text/javascript application/json image/svg+xml text/xml application/xml
</IfModule>
